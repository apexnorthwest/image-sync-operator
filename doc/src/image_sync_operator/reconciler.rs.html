14/// Checks if the spec of the ImageSync CR has changed since the last time it was applied. Returns true if it has changed, false otherwise.
15///
16/// All this function does is convert the spec and status.last_applied_config to JSON and compare them. If they are equal, then the spec has not changed. If they are not equal, then the spec has changed.
17pub async fn has_config_changed(obj: &ImageSync) -> Result<bool, Box<dyn std::error::Error>> {
18    let spec_json = serde_json::to_value(obj.spec.clone())?;
19    if let Some(status) = &obj.status
20        && let Some(last_applied_config) = Some(status.clone().last_applied_config)
21    {
22        let last_applied_config_json =
23            serde_json::to_value(serde_json::from_str::<ImageSyncSpec>(&last_applied_config)?)?;
24        if last_applied_config_json.eq(&spec_json) {
25            return Ok(false);
26        }
27    }
28    Ok(true)
29}

31/// Take a list of new status values and update the CR. This is a common task in the reconciler loop so we abstract it out.
32pub async fn update_status(
33    obj: ImageSync,
34    accepted: bool,
35    ready: bool,
36    accepted_message: String,
37    last_completion_time: Option<Time>,
38    _client: &Client,
39) -> Result<(), Box<dyn std::error::Error>> {
40    let imagesyncs: Api<ImageSync> = Api::namespaced(
41        kube::Client::try_default().await.unwrap(),
42        obj.metadata.namespace.as_ref().unwrap().as_str(),
43    );
44    println!(
45        "Updating status for ImageSync {} in namespace {}: accepted={}, ready={}, message={}",
46        obj.metadata.name.as_ref().unwrap(),
47        obj.metadata.namespace.as_ref().unwrap(),
48        accepted,
49        ready,
50        accepted_message
51    );
52    let mut patched_obj = imagesyncs
53        .get(obj.metadata.name.as_ref().unwrap().as_str())
54        .await?;
55    patched_obj.status = Some(ImageSyncStatus {
56        accepted,
57        message: accepted_message,
58        ready,
59        last_applied_config: serde_json::to_string(&obj.spec)?,
60        last_completion_time,
61    });
62    let post_params = PostParams::default();
63    match imagesyncs
64        .replace_status(
65            patched_obj.metadata.name.as_ref().unwrap().as_str(),
66            &post_params,
67            &patched_obj,
68        )
69        .await
70    {
71        Ok(_) => {
72            println!(
73                "Successfully updated status for ImageSync {}",
74                obj.metadata.name.as_ref().unwrap()
75            );
76            Ok(())
77        }
78        Err(e) => {
79            println!(
80                "Failed to update status for ImageSync {}: {}",
81                obj.metadata.name.as_ref().unwrap(),
82                e
83            );
84            println!("{}", serde_json::to_string_pretty(&patched_obj).unwrap());
85            Err(Box::new(e))
86        }
87    }
88}

90/// When the ImageSync spec has changed, we need to reset the status to not accepted.
91///
92/// This function simply calls the update_status function with the appropriate values to reset the status to not accepted.
93pub async fn reset_to_not_accepted(
94    obj: &ImageSync,
95    client: &Client,
96) -> Result<(), Box<dyn std::error::Error>> {
97    println!(
98        "Resetting ImageSync {} to not accepted",
99        obj.metadata.name.clone().unwrap()
100    );
101    let basename = if obj.metadata.name.as_ref().unwrap().len() > 50 {
102        obj.metadata.name.as_ref().unwrap()[0..50].to_string()
103    } else {
104        obj.metadata.name.as_ref().unwrap().to_string()
105    };
106    let jobs = Api::<Job>::namespaced(client.clone(), obj.metadata.namespace.as_ref().unwrap());
107    let cronjobs = Api::<k8s_openapi::api::batch::v1::CronJob>::namespaced(
108        client.clone(),
109        obj.metadata.namespace.as_ref().unwrap(),
110    );
111    if jobs.get(obj.metadata.name.as_ref().unwrap()).await.is_ok() {
112        println!(
113            "Deleting job imagesync-{} because the spec has changed",
114            basename
115        );
116        jobs.delete(
117            format!("imagesync-{}", basename).as_str(),
118            &Default::default(),
119        )
120        .await?;
121    }
122    if cronjobs
123        .get(obj.metadata.name.as_ref().unwrap())
124        .await
125        .is_ok()
126    {
127        println!(
128            "Deleting cronjob imagesync-{} because the spec has changed",
129            basename
130        );
131        cronjobs
132            .delete(
133                format!("imagesync-{}", basename).as_str(),
134                &Default::default(),
135            )
136            .await?;
137    }
138    update_status(
139        obj.clone(),
140        false,
141        false,
142        String::from("ImageSync configuration has changed"),
143        None,
144        client,
145    )
146    .await?;
147    Ok(())
148}

150/// Run acceptance checks on the ImageSync spec.
151///
152/// Whenever the spec changes, we need to verify that it's valid and that the secrets exist.
153/// This function returns true if the spec is valid and false otherwise.
154/// It also calls the update_status function to update the status of the ImageSync CR with the results of the acceptance checks.
155pub async fn acceptance_checks(
156    obj: &ImageSync,
157    client: &Client,
158) -> Result<bool, Box<dyn std::error::Error>> {
159    let mut source_secret_okay = true;
160    let mut source_secret_message = String::new();
161    let mut dest_secret_okay = true;
162    let mut dest_secret_message = String::new();
163    let mut cron_schedule_okay = true;
164    let mut source_image_okay = true;
165    let mut dest_image_okay = true;
166
167    let secrets = Api::<k8s_openapi::api::core::v1::Secret>::namespaced(
168        client.clone(),
169        obj.metadata.namespace.as_ref().unwrap().as_str(),
170    );
171
172    // Acceptance check for the source secret
173    if let Some(secret_name) = &obj.spec.source.registry_login_secret {
174        match secrets.get(secret_name).await {
175            Ok(secret) => {
176                println!("Source secret {} exists", secret_name);
177                if secret.type_ != Some(String::from("kubernetes.io/dockerconfigjson")) {
178                    println!(
179                        "Source secret {} is not of type kubernetes.io/dockerconfigjson",
180                        secret_name
181                    );
182                    source_secret_okay = false;
183                    source_secret_message = format!(
184                        "Source secret {} is not of type kubernetes.io/dockerconfigjson",
185                        secret_name
186                    );
187                }
188            }
189            Err(e) => {
190                println!("Source secret {} does not exist: {}", secret_name, e);
191                source_secret_okay = false;
192                source_secret_message = format!("Source secret {} does not exist", secret_name);
193            }
194        }
195    }
196
197    // Acceptance check for the destination secret
198    if let Some(secret_name) = &obj.spec.destination.registry_login_secret {
199        match secrets.get(secret_name).await {
200            Ok(secret) => {
201                println!("Destination secret {} exists", secret_name);
202                if secret.type_ != Some(String::from("kubernetes.io/dockerconfigjson")) {
203                    println!(
204                        "Destination secret {} is not of type kubernetes.io/dockerconfigjson",
205                        secret_name
206                    );
207                    dest_secret_okay = false;
208                    dest_secret_message = format!(
209                        "Destination secret {} is not of type kubernetes.io/dockerconfigjson",
210                        secret_name
211                    );
212                }
213            }
214            Err(e) => {
215                println!("Destination secret {} does not exist: {}", secret_name, e);
216                dest_secret_okay = false;
217                dest_secret_message = format!("Destination secret {} does not exist", secret_name);
218            }
219        }
220    }
221
222    // Acceptance check for the cron schedule
223    if let Some(schedule) = &obj.spec.cron_schedule {
224        // This regex is a bit of a guess since the official k8s api spec doesn't define the exact format. We presume it's the same as most other cron implementions.
225        if Regex::new(r"^((((\d+,)+\d+|(\d+(/|-|#)\d+)|\d+L?|\*(/\d+)?|L(-\d+)?|\?|[A-Z]{3}(-[A-Z]{3})?) ?){5,7})$|(@(annually|yearly|monthly|weekly|daily|hourly|reboot))|(@every (\d+(ns|us|µs|ms|s|m|h))+)$").unwrap().is_match(schedule) {
226            println!("Cron schedule {} is valid", schedule);
227        } else {
228            println!("Cron schedule {} is invalid", schedule);
229            cron_schedule_okay = false;
230        }
231    }
232
233    // Acceptance check for the source image URL
234    if !Regex::new(r"^(([a-zA-Z0-9]|[a-zA-Z0-9][a-zA-Z0-9\-]*[a-zA-Z0-9])\.)*([A-Za-z0-9]|[A-Za-z0-9][A-Za-z0-9\-]*[A-Za-z0-9])(:[0-9]+)?/[a-zA-Z0-9_\-\.]+/[a-zA-Z0-9_\-\.]+(:[a-zA-Z0-9_\-\.]{1,128}|@sha256:[a-f0-9]+)").unwrap().is_match(&obj.spec.source.image) {
235        println!("Source image URL {} is invalid", obj.spec.source.image);
236        source_image_okay = false;
237    }
238
239    // Acceptance check for the destination image URL
240    if !Regex::new(r"^(([a-zA-Z0-9]|[a-zA-Z0-9][a-zA-Z0-9\-]*[a-zA-Z0-9])\.)*([A-Za-z0-9]|[A-Za-z0-9][A-Za-z0-9\-]*[A-Za-z0-9])(:[0-9]+)?/[a-zA-Z0-9_\-\.]+/[a-zA-Z0-9_\-\.]+(:[a-zA-Z0-9_\-\.]{1,128})").unwrap().is_match(&obj.spec.destination.image) {
241        println!("Destination image URL {} is invalid", obj.spec.destination.image);
242        dest_image_okay = false;
243    }
244
245    // Determine if the ImageSync configuration is accepted and render the message accordingly
246    let mut accepted = true;
247    let mut accepted_message = String::from("ImageSync configuration is valid");
248    if !source_secret_okay
249        || !dest_secret_okay
250        || !cron_schedule_okay
251        || !source_image_okay
252        || !dest_image_okay
253    {
254        accepted = false;
255        accepted_message = String::from("ImageSync configuration is invalid: ");
256        if !source_secret_okay {
257            accepted_message.push_str(&format!("Source secret error: {}. ", source_secret_message));
258        }
259        if !dest_secret_okay {
260            accepted_message.push_str(&format!(
261                "Destination secret error: {}. ",
262                dest_secret_message
263            ));
264        }
265        if !cron_schedule_okay {
266            accepted_message.push_str("Cron schedule is invalid. ");
267        }
268        if !source_image_okay {
269            accepted_message.push_str("Source image URL is invalid. ");
270        }
271        if !dest_image_okay {
272            accepted_message.push_str("Destination image URL is invalid. ");
273        }
274    }
275
276    update_status(obj.clone(), accepted, false, accepted_message, None, client).await?;
277    Ok(accepted)
278}
