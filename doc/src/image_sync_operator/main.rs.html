3//! image-sync-operator is a Kubernetes operator that synchronizes container images between registries based on the ImageSync custom resource definition (CRD).
4//! 
5//! Installing the operator is typically done via helm by pulling the chart from the public oci repository.
6//! You can also find the charts in the legacy helm registry format [here](https://apexnorthwest.github.io/image-sync-operator/charts/).
7//! 
8//! To install the operator in single-namespace mode with the default settings you would do the following:
9//! ```sh
10//! helm repo add apexnw oci://ghcr.io/apexnorthwest/charts
11//! helm install image-sync-operator apexnw/image-sync-operator -n image-sync-operator --create-namespace
12//! ```
13//! 
14//! To fully customize the configuration you would pass a values.yaml files like so:
15//! ```yaml
16//! ---
17//! # Settings for the operator itself. This is the container that runs the operator code and watches for ImageSync CRs.
18//! operator:
19//!   # Image url to pull from
20//!   image: 
21//!     repository: "ghcr.io/apexnorthwest/image-sync-operator"
22//!     tag: "0.1.0"
23//!     pullPolicy: "IfNotPresent"
24//!   # Enable for debug logging. This is not recommended for production use but if you even need to open an issue
25//!   # on the project then we will expect you to provide logs with this enabled.
26//!   debug: false
27//! service_account:
28//!   # Should this chart create a service account?
29//!   # If false, the service account named below must already exist
30//!   create: true
31//!   # Name of the service account to use/create
32//!   # Defaults to release name if not set
33//!   name: ""
34//! # Settings related to the skopeo image used by the operator
35//! # Technically, any image that contains skopeo can be used, but we prefer the official one
36//! # Notably, the ca trust must be in /etc/pki/ca-trust/extracted/pem/tls-ca-bundle.pem, so
37//! # if you use an image based on a different OS, it's likely it will have tls problems.
38//! skopeo:
39//!   # Image url to pull from
40//!   image:
41//!     repository: "quay.io/skopeo/stable"
42//!     tag: "latest"
43//!     pullPolicy: "IfNotPresent"
44//!   # Optional CA trust bundle to use when verifying TLS connections to registries
45//!   # This should be a PEM encoded bundle of CA certificates, and will be appended
46//!   # to the system CA trust bundle in the skopeo container. This is generally mandatory when
47//!   # working with private, authenticated registries.
48//!   ca_trust_bundle: |-
49//!     -----BEGIN CERTIFICATE-----
50//!     ***********omitted***********
51//!     -----END CERTIFICATE-----
52//! # Settings related to the operator's access
53//! rbac:
54//!   # Should this chart create the cluster role `image-sync-operator`?
55//!   # This must be set to true if cluster_scoped is true, otherwise it is optional.
56//!   create_cluster_role: false
57//!   # Will this operator need to be cluster scoped?
58//!   # If true, the operator will watch all namespaces and require cluster role permissions.
59//!   cluster_scoped: false
60//!   # When not in cluster scoped mode, which namespaces should the operator watch?
61//!   # If empty, the operator will watch the namespace it is deployed in only.
62//!   # This will be ignored if cluster_scoped is true.
63//!   # The operator service account will need to have a rolebinding in these namespaces to be able to watch for ImageSync CRs.
64//!   # The Role can be copied from the install namespace, or if create_cluster_role is true then you can bind that cluster role into the namespace.
65//!   watched_namespaces:
66//!     - "image-sync-operator"
67//!     - "default"
68//!     - "some-other-application"
69//! ```
70//! 
71//! You would then create ImageSync CRs in that same namespace (as in this example we've installed the operator in restricted mode, as is the default)
72//! 
73//! The following is an example of an ImageSync:
74//! ```yaml
75//! ---
76//! apiVersion: imagesync.apexnw.dev/v1alpha1
77//! kind: ImageSync
78//! metadata:
79//!   name: alpine-latest
80//!   namespace: image-sync-operator
81//! spec:
82//!   source:
83//!     # Full URI of the source image as would be passed to skopeo (minus the docker:// prefix, which is presumed)
84//!     # Note: You cannot use docker short names for images. You must use full docker.io paths, as skopeo cannot know what registry to use for short names. This applies to both source and dest.
85//!     # This can also be an @ hash reference. Destination however, cannot be as you must tag the pushed image to something.
86//!     image: "docker.io/library/alpine:latest"
87//!     # Optional: The name of a Secret in the same namespace of type kubernetes.io/dockerconfigjson used to authenticate to the source registry.
88//!     registryLoginSecret: "dockerhub"
89//!   destination:
90//!     # Full URI of the destination image as would be passed to skopeo (minus the docker:// prefix, which is presumed)
91//!     image: "my-private-registry.example.com/library/alpine:latest"
92//!     # Optional: The name of a Secret in the same namespace of type kubernetes.io/dockerconfigjson used to authenticate to the destination registry.
93//!     registryLoginSecret: "private-registry"
94//!   # Optional: If you want to rerun the sync on a schedule after the initial sync. Uses CronJob format. If unset, the sync only runs once. 
95//!   cronSchedule: "*/15 * * * *"
96//!   # Optional: If true, all architectures will be copied. If false, only the architecture of the operator pod will be copied. Defaults to false. This is the same as passing --all to skopeo.
97//!   allArchitectures: true
98//!   # Optional: If true, the destination image will be tagged with the digest of the source image. If false, the destination image will be tagged with the tag of the source image. Defaults to false. This is the same as passing --preserve-digests to skopeo
99//!   # Notably, if this is set to true, and allArchitectures is set to false, multi-arch images will fail to copy and the sync will fail.
100//!   preserveDigests: true
101//!   # Optional: If you want to pass extra arguments to skopeo, you can do so here. This is a string that will be split on whitespace and passed to skopeo as-is. For example, if you want to pass --insecure-policy to skopeo, you would set this to "--insecure-policy".
102//!   # Note that this is not a list of arguments, but a single string that will be split on whitespace. If it needs to have spaces in an argument (which it shouldn't), you will need to use single quotes inside the string.
103//!   extraSkopeoArguments: ""
104//! ```
105//! 
106//! This project is not meant to be used as a library or installed via cargo. As a result, all contained functions and modules documented within this site are 
107//! written for the use of contributors and not users of the operator.

146/// Entrypoint for the operator. This main function sets up the leader election, watches for ImageSync CRD changes,
147/// and calls the reconciler when a change is detected. It uses tokio as the async runtime and is not meant to be called directly.
148#[tokio::main]
149async fn main() -> Result<(), kube::Error> {
150    let client = Client::try_default().await?;
151    let namespace = client.default_namespace();
152
153    let manager_client = Client::try_default().await?;
154    let manager = LeaseManagerBuilder::new(manager_client, "image-sync-operator")
155        .with_namespace(namespace)
156        .build()
157        .await
158        .unwrap();
159
160    let controller_client = Client::try_default().await?;
161    let mut imagesyncs: Vec<Api<ImageSync>>;
162    if CONFIG.cluster_scope {
163        println!("Operator is running in cluster-scoped mode");
164        imagesyncs = vec![Api::<ImageSync>::all(controller_client)];
165    } else {
166        println!("Operator is running in namespace-scoped mode");
167        imagesyncs = Vec::<Api<ImageSync>>::new();
168        let namespaces = if !CONFIG.watched_namespaces.contains(&namespace.to_string()) {
169            vec![namespace.to_string()]
170        } else {
171            CONFIG.watched_namespaces.clone()
172        };
173        for ns in namespaces.iter() {
174            let api = Api::<ImageSync>::namespaced(controller_client.clone(), ns);
175            imagesyncs.push(api);
176        }
177    }
178    println!("Operator has {} watchers", imagesyncs.len());
179    println!(
180        "Operator is starting with the following configuration: {:?}",
181        *CONFIG
182    );
183
184    // Start manager in watching mode and get back status channel and task handler.
185    let (mut channel, task) = manager.watch().await;
186
187    // Watch on the channel for lock state changes.
188    tokio::select! {
189        _ = channel.changed() => {
190            let lock_state = *channel.borrow_and_update();
191
192            if lock_state {
193                // Do something useful as a leader
194                println!("Operator has become the leader");
195                if CONFIG.cluster_scope {
196                    println!("Starting controller for cluster-scoped ImageSyncs");
197                    Controller::new(imagesyncs[0].clone(), Default::default())
198                        .run(reconcile, error_policy, Arc::new(()))
199                        .for_each(|_| futures::future::ready(())).await;
200                } else {
201                    for ns in imagesyncs.iter() {
202                        println!("Starting controller for namespace: {}", ns.namespace().unwrap());
203                        Controller::new(ns.clone(), Default::default())
204                            .run(reconcile, error_policy, Arc::new(()))
205                            .for_each(|_| futures::future::ready(())).await;
206                    }
207                }
208            }
209        }
210        _ = tokio::time::sleep(Duration::from_secs(60)) => {
211            println!("Unable to get lock during 60s");
212        }
213    }
214    println!("Operator is shutting down");
215
216    // Explicitly close the control channel
217    drop(channel);
218
219    // Wait for the finish of the manager and get it back
220    let _manager = tokio::join!(task).0.unwrap();
221
222    Ok(())
223}

225/// This reconcile funtion is the main logical driver of the operator. It is called whenever an event is generated related to an ImageSync CRD.
226/// This runs as a single listener in global or single-namespace mode. In multi-namespace mode it will run as a listener on each namespace it monitors.
227/// Notably, this function usually only performs a single action when called, even if multiple actions are called for. This is because we want the function
228/// to be as idempotent as possible. It also ensures that no matter what state the ImageSync is in, it will always progress towards the correct state.
229/// 
230/// This function is not meant to be called from anywhere other than the controller thread.
231async fn reconcile(obj: Arc<ImageSync>, _ctx: Arc<()>) -> Result<Action> {
232    // Configure K8s api connection
233    let client = Client::try_default().await.unwrap();
234
235    // Check if the config has changed or is new
236    if reconciler::has_config_changed(&obj).await.unwrap() {
237        // Spec has changed, reset status
238        reconciler::reset_to_not_accepted(&obj, &client)
239            .await
240            .unwrap();
241        // Run acceptance checks on the new spec and update status accordingly
242        if reconciler::acceptance_checks(&obj, &client).await.unwrap() {
243            // Immediate requeue to process the new spec
244            return Ok(Action::requeue(Duration::ZERO));
245        } else {
246            // Longer requeue due to acceptance failure
247            return Ok(Action::requeue(Duration::from_secs(30)));
248        }
249    }
250
251    // Check if we are Accepted or not
252    if !obj.status.as_ref().unwrap().accepted {
253        // Run acceptance checks
254        if reconciler::acceptance_checks(&obj, &client).await.unwrap() {
255            // Immediate requeue to process the new spec
256            return Ok(Action::requeue(Duration::ZERO));
257        } else {
258            // Longer requeue due to acceptance failure
259            return Ok(Action::requeue(Duration::from_secs(30)));
260        }
261    }
262
263    // Process the CR when it's both accepted and ready
264    if obj.status.as_ref().unwrap().ready {
265        if obj.spec.cron_schedule.is_some() {
266            // TODO: Implement cron job features
267        } else {
268            // If the CR is ready and does not have a cron schedule, there's nothing to do. Do a very long requeue.
269            return Ok(Action::requeue(Duration::from_secs(3600)));
270        }
271    }
272
273    // If the CR is accepted but not ready, we need to create a one-shot job to process it. We always fire a one-shot, even when the job is scheduled.
274    let job = jobs::get_job_for_imagesync(&obj, &client).await.unwrap();
275
276    // Create job when absent
277    if job.is_none() {
278        jobs::create_job(&obj, &CONFIG.skopeo, &client)
279            .await
280            .unwrap();
281        // Fast requeue to watch the job status
282        return Ok(Action::requeue(Duration::from_secs(5)));
283    }
284
285    // If we get here, the job exists. Check its status.
286    if jobs::is_job_complete(job.as_ref().unwrap()).await {
287        // Job is not running, check if it succeeded or failed
288        if jobs::is_job_failed(job.as_ref().unwrap()).await {
289            // Job failed. Check how long ago the job failed. If it was more than 10 minutes ago we will delete it to retry.
290            let last_completion_time = job
291                .as_ref()
292                .unwrap()
293                .status
294                .as_ref()
295                .unwrap()
296                .completion_time
297                .clone()
298                .unwrap();
299            if last_completion_time
300                .0
301                .duration_until(Timestamp::now())
302                .as_mins()
303                > 10
304            {
305                // Delete the job to retry
306                jobs::delete_job(job.as_ref().unwrap(), &client)
307                    .await
308                    .unwrap();
309                // Fast requeue to watch the job status
310                reconciler::update_status(
311                    obj.as_ref().clone(),
312                    true,
313                    false,
314                    String::from("ImageSync job failed, retrying"),
315                    None,
316                    &client,
317                )
318                .await
319                .unwrap();
320                return Ok(Action::requeue(Duration::from_secs(5)));
321            } else {
322                reconciler::update_status(
323                    obj.as_ref().clone(),
324                    true,
325                    false,
326                    String::from("ImageSync job failed"),
327                    None,
328                    &client,
329                )
330                .await
331                .unwrap();
332            }
333        } else {
334            // Job succeeded. Update the status to ready.
335            reconciler::update_status(
336                obj.as_ref().clone(),
337                true,
338                true,
339                String::from("ImageSync job completed successfully"),
340                Some(
341                    job.as_ref()
342                        .unwrap()
343                        .status
344                        .as_ref()
345                        .unwrap()
346                        .completion_time
347                        .clone()
348                        .unwrap(),
349                ),
350                &client,
351            )
352            .await
353            .unwrap();
354            // Fast requeue since we don't know if we need to create a CronJob or not.
355            return Ok(Action::requeue(Duration::from_secs(5)));
356        }
357    } else {
358        // Job is still running, update the status to not ready.
359        reconciler::update_status(
360            obj.as_ref().clone(),
361            true,
362            false,
363            String::from("ImageSync job is running"),
364            None,
365            &client,
366        )
367        .await
368        .unwrap();
369        // Fast requeue to watch the job status
370        return Ok(Action::requeue(Duration::from_secs(5)));
371    }
372
373    // Fallback requeue for periodic check (5min)
374    Ok(Action::requeue(Duration::from_secs(300)))
375}

377/// This function is called when the reconciler encounters an error. In this case, we simply retry the reconcile after 30 seconds.
378/// Because the reconciler is idempotent, this is safe to do. The error is logged by the controller runtime.
379fn error_policy(_object: Arc<ImageSync>, _err: &Error, _ctx: Arc<()>) -> Action {
380    eprintln!("Reconcile error: {:?} on {:?}", _err, _object.metadata.name);
381    Action::requeue(Duration::from_secs(30))
