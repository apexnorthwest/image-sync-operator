3//! image-sync-operator is a Kubernetes operator that synchronizes container images between registries based on the ImageSync custom resource definition (CRD).
4//!
5//! Installing the operator is typically done via helm by pulling the chart from the public oci repository.
6//! You can also find the charts in the legacy helm registry format [here](https://apexnorthwest.github.io/image-sync-operator/charts/).
7//!
8//! To install the operator in single-namespace mode with the default settings you would do the following:
9//! ```sh
10//! helm repo add apexnw oci://ghcr.io/apexnorthwest/charts
11//! helm install image-sync-operator apexnw/image-sync-operator -n image-sync-operator --create-namespace
12//! ```
13//!
14//! To fully customize the configuration you would pass a values.yaml files like so:
15//! ```yaml
16//! ---
17//! # Settings for the operator itself. This is the container that runs the operator code and watches for ImageSync CRs.
18//! operator:
19//!   # Image url to pull from
20//!   image:
21//!     repository: "ghcr.io/apexnorthwest/image-sync-operator"
22//!     tag: "0.1.0"
23//!     pullPolicy: "IfNotPresent"
24//!   # Enable for debug logging. This is not recommended for production use but if you even need to open an issue
25//!   # on the project then we will expect you to provide logs with this enabled.
26//!   debug: false
27//! service_account:
28//!   # Should this chart create a service account?
29//!   # If false, the service account named below must already exist
30//!   create: true
31//!   # Name of the service account to use/create
32//!   # Defaults to release name if not set
33//!   name: ""
34//! # Settings related to the skopeo image used by the operator
35//! # Technically, any image that contains skopeo can be used, but we prefer the official one
36//! # Notably, the ca trust must be in /etc/pki/ca-trust/extracted/pem/tls-ca-bundle.pem, so
37//! # if you use an image based on a different OS, it's likely it will have tls problems.
38//! skopeo:
39//!   # Image url to pull from
40//!   image:
41//!     repository: "quay.io/skopeo/stable"
42//!     tag: "latest"
43//!     pullPolicy: "IfNotPresent"
44//!   # Optional CA trust bundle to use when verifying TLS connections to registries
45//!   # This should be a PEM encoded bundle of CA certificates, and will be appended
46//!   # to the system CA trust bundle in the skopeo container. This is generally mandatory when
47//!   # working with private, authenticated registries.
48//!   ca_trust_bundle: |-
49//!     -----BEGIN CERTIFICATE-----
50//!     ***********omitted***********
51//!     -----END CERTIFICATE-----
52//! # Settings related to the operator's access
53//! rbac:
54//!   # Should this chart create the cluster role `image-sync-operator`?
55//!   # This must be set to true if cluster_scoped is true, otherwise it is optional.
56//!   create_cluster_role: false
57//!   # Will this operator need to be cluster scoped?
58//!   # If true, the operator will watch all namespaces and require cluster role permissions.
59//!   cluster_scoped: false
60//!   # When not in cluster scoped mode, which namespaces should the operator watch?
61//!   # If empty, the operator will watch the namespace it is deployed in only.
62//!   # This will be ignored if cluster_scoped is true.
63//!   # The operator service account will need to have a rolebinding in these namespaces to be able to watch for ImageSync CRs.
64//!   # The Role can be copied from the install namespace, or if create_cluster_role is true then you can bind that cluster role into the namespace.
65//!   watched_namespaces:
66//!     - "image-sync-operator"
67//!     - "default"
68//!     - "some-other-application"
69//! ```
70//!
71//! You would then create ImageSync CRs in that same namespace (as in this example we've installed the operator in restricted mode, as is the default)
72//!
73//! The following is an example of an ImageSync:
74//! ```yaml
75//! ---
76//! apiVersion: imagesync.apexnw.dev/v1alpha1
77//! kind: ImageSync
78//! metadata:
79//!   name: alpine-latest
80//!   namespace: image-sync-operator
81//! spec:
82//!   source:
83//!     # Full URI of the source image as would be passed to skopeo (minus the docker:// prefix, which is presumed)
84//!     # Note: You cannot use docker short names for images. You must use full docker.io paths, as skopeo cannot know what registry to use for short names. This applies to both source and dest.
85//!     # This can also be an @ hash reference. Destination however, cannot be as you must tag the pushed image to something.
86//!     image: "docker.io/library/alpine:latest"
87//!     # Optional: The name of a Secret in the same namespace of type kubernetes.io/dockerconfigjson used to authenticate to the source registry.
88//!     registryLoginSecret: "dockerhub"
89//!   destination:
90//!     # Full URI of the destination image as would be passed to skopeo (minus the docker:// prefix, which is presumed)
91//!     image: "my-private-registry.example.com/library/alpine:latest"
92//!     # Optional: The name of a Secret in the same namespace of type kubernetes.io/dockerconfigjson used to authenticate to the destination registry.
93//!     registryLoginSecret: "private-registry"
94//!   # Optional: If you want to rerun the sync on a schedule after the initial sync. Uses CronJob format. If unset, the sync only runs once.
95//!   cronSchedule: "*/15 * * * *"
96//!   # Optional: If true, all architectures will be copied. If false, only the architecture of the operator pod will be copied. Defaults to false. This is the same as passing --all to skopeo.
97//!   allArchitectures: true
98//!   # Optional: If true, the destination image will be tagged with the digest of the source image. If false, the destination image will be tagged with the tag of the source image. Defaults to false. This is the same as passing --preserve-digests to skopeo
99//!   # Notably, if this is set to true, and allArchitectures is set to false, multi-arch images will fail to copy and the sync will fail.
100//!   preserveDigests: true
101//!   # Optional: If you want to pass extra arguments to skopeo, you can do so here. This is a string that will be split on whitespace and passed to skopeo as-is. For example, if you want to pass --insecure-policy to skopeo, you would set this to "--insecure-policy".
102//!   # Note that this is not a list of arguments, but a single string that will be split on whitespace. If it needs to have spaces in an argument (which it shouldn't), you will need to use single quotes inside the string.
103//!   extraSkopeoArguments: ""
104//! ```
105//!
106//! This project is not meant to be used as a library or installed via cargo. As a result, all contained functions and modules documented within this site are
107//! written for the use of contributors and not users of the operator.

144/// Entrypoint for the operator. This main function sets up the leader election, watches for ImageSync CRD changes,
145/// and calls the reconciler when a change is detected. It uses tokio as the async runtime and is not meant to be called directly.
146#[tokio::main]
147async fn main() -> Result<(), kube::Error> {
148    // Use unwrap during initialization of the operator because all of these errors are fatal anyways.
149    // At some point we should handle them if for no other reason than to print better error messages.
150    aws_lc_rs::default_provider().install_default().unwrap();
151
152    let client = Client::try_default().await?;
153    let namespace = client.default_namespace();
154
155    let manager_client = Client::try_default().await?;
156    let manager = LeaseManagerBuilder::new(manager_client, "image-sync-operator")
157        .with_namespace(namespace)
158        .build()
159        .await
160        .unwrap();
161
162    let controller_client = Client::try_default().await?;
163    let mut imagesyncs: Vec<Api<ImageSync>>;
164    if CONFIG.cluster_scope {
165        println!("Operator is running in cluster-scoped mode");
166        imagesyncs = vec![Api::<ImageSync>::all(controller_client)];
167    } else {
168        println!("Operator is running in namespace-scoped mode");
169        imagesyncs = Vec::<Api<ImageSync>>::new();
170        let namespaces = if !CONFIG.watched_namespaces.contains(&namespace.to_string()) {
171            vec![namespace.to_string()]
172        } else {
173            CONFIG.watched_namespaces.clone()
174        };
175        for ns in namespaces.iter() {
176            let api = Api::<ImageSync>::namespaced(controller_client.clone(), ns);
177            imagesyncs.push(api);
178        }
179    }
180    println!("Operator has {} watchers", imagesyncs.len());
181    println!(
182        "Operator is starting with the following configuration: {:?}",
183        *CONFIG
184    );
185
186    // Start manager in watching mode and get back status channel and task handler.
187    let (mut channel, task) = manager.watch().await;
188
189    // Watch on the channel for lock state changes.
190    tokio::select! {
191        _ = channel.changed() => {
192            let lock_state = *channel.borrow_and_update();
193
194            if lock_state {
195                // Do something useful as a leader
196                println!("Operator has become the leader");
197                if CONFIG.cluster_scope {
198                    println!("Starting controller for cluster-scoped ImageSyncs");
199                    Controller::new(imagesyncs[0].clone(), Default::default())
200                        .run(reconcile, error_policy, Arc::new(()))
201                        .for_each(|item| async {
202                            // We expect the loop to never exist, as the controller should run indefinitely
203                            // There are some errors that we expect to be able to safely ignore, some indicate
204                            // issues talking to the apiserver and on those or any unknown errors we want to panic out the process.
205                            match item {
206                                Ok(_) => {}
207                                Err(kube::runtime::controller::Error::ObjectNotFound(_)) => {}
208                                Err(kube::runtime::controller::Error::ReconcilerFailed(..)) => {}
209                                Err(err) => panic!("ImageSync controller infrastructure failed: {err:?}"),
210                            }
211                        })
212                        .await;
213                    // No shutdown path exists in this operator, so stream completion is abnormal
214                    panic!("cluster-scoped ImageSync controller stream terminated");
215                } else {
216                    let controllers = imagesyncs.iter().map(|ns| {
217                        let ns_name = ns.namespace().unwrap().to_string();
218                        println!("Starting controller for namespace: {ns_name}");
219                        let stream = Controller::new(ns.clone(), Default::default())
220                            .run(reconcile, error_policy, Arc::new(()));
221                        async move {
222                            let controller_ns_name = ns_name.clone();
223                            stream
224                                .for_each(move |item| {
225                                    let ns_name = controller_ns_name.clone();
226                                    async move {
227                                        // As mentioned above, we expect some errors but others are fatal.
228                                        match item {
229                                            Ok(_) => {}
230                                            Err(kube::runtime::controller::Error::ObjectNotFound(_)) => {}
231                                            Err(kube::runtime::controller::Error::ReconcilerFailed(..)) => {}
232                                            Err(err) => panic!("ImageSync controller in {ns_name} failed: {err:?}"),
233                                        }
234                                    }
235                                })
236                                .await;
237                            panic!("ImageSync controller watching {ns_name} terminated");
238                        }
239                    });
240                    futures::future::join_all(controllers).await;
241                }
242            }
243        }
244        _ = tokio::time::sleep(Duration::from_secs(60)) => {
245            println!("Unable to get lock during 60s");
246        }
247    }
248    println!("Operator is shutting down");
249
250    // Explicitly close the control channel
251    drop(channel);
252
253    // Wait for the finish of the manager and get it back
254    let _manager = tokio::join!(task).0.unwrap();
255
256    Ok(())
257}

259/// This reconcile funtion is the main logical driver of the operator. It is called whenever an event is generated related to an ImageSync CRD.
260/// This runs as a single listener in global or single-namespace mode. In multi-namespace mode it will run as a listener on each namespace it monitors.
261/// Notably, this function usually only performs a single action when called, even if multiple actions are called for. This is because we want the function
262/// to be as idempotent as possible. It also ensures that no matter what state the ImageSync is in, it will always progress towards the correct state.
263///
264/// This function is not meant to be called from anywhere other than the controller thread.
265async fn reconcile(obj: Arc<ImageSync>, _ctx: Arc<()>) -> Result<Action> {
266    // Init local variables to reduce the number of times we have to access the Arc
267    let namespace = obj
268        .metadata
269        .namespace
270        .as_ref()
271        .ok_or_else(|| OpError::from("ImageSync is missing its namespace".to_string()))?
272        .clone();
273    let name = obj
274        .metadata
275        .name
276        .as_ref()
277        .ok_or_else(|| OpError::from("ImageSync is missing its name".to_string()))?
278        .clone();
279
280    // Configure K8s api connection
281    let client = Client::try_default().await?;
282
283    let imagesyncs: Api<ImageSync> = Api::namespaced(client.clone(), &namespace);
284
285    // Check if the imagesync still exists
286    match imagesyncs.get(&name).await {
287        Ok(_) => {
288            // ImageSync still exists, continue processing
289        }
290        Err(kube::Error::Api(e)) if e.code == 404 => {
291            println!(
292                "ImageSync {} in namespace {} has been deleted. Cleaning up.",
293                name, namespace
294            );
295            // Delete any jobs or cronjobs that might exist for this ImageSync
296            let job = jobs::get_job_for_imagesync(&obj, &client).await?;
297            if let Some(job) = job {
298                jobs::delete_job(&job, &client).await?;
299            }
300            let cronjob = cronjobs::get_cronjob_for_imagesync(&obj, &client).await?;
301            if let Some(cronjob) = cronjob {
302                cronjobs::delete_cronjob(&cronjob, &client).await?;
303            }
304            return Ok(Action::await_change());
305        }
306        Err(e) => {
307            eprintln!(
308                "Error getting ImageSync {} in namespace {}: {}",
309                name, namespace, e
310            );
311            return Err(e.into());
312        }
313    }
314
315    // Check if the config has changed or is new
316    if reconciler::has_config_changed(&obj).await? {
317        println!(
318            "Spec of ImageSync {} in namespace {} has changed. Resetting status and deleting any jobs or cronjobs.",
319            name, namespace
320        );
321        // Spec has changed, reset status and delete any job or cronjob that might exist
322        reconciler::reset_to_not_accepted(&obj, &client).await?;
323        // Run acceptance checks on the new spec and update status accordingly
324        if reconciler::acceptance_checks(&obj, &client).await? {
325            // Immediate requeue to process the new spec
326            return Ok(Action::requeue(Duration::ZERO));
327        } else {
328            // Longer requeue due to acceptance failure
329            return Ok(Action::requeue(Duration::from_secs(30)));
330        }
331    }
332
333    // Check if we are Accepted or not
334    let status = obj.status.as_ref().ok_or_else(|| {
335        OpError::from(format!(
336            "ImageSync {name} has no status after configuration validation"
337        ))
338    })?;
339    if !status.accepted {
340        // Run acceptance checks
341        if reconciler::acceptance_checks(&obj, &client).await? {
342            // Immediate requeue to process the new spec
343            return Ok(Action::requeue(Duration::ZERO));
344        } else {
345            // Longer requeue due to acceptance failure
346            return Ok(Action::requeue(Duration::from_secs(30)));
347        }
348    }
349
350    // Process the CR when it's both accepted and ready
351    if status.ready {
352        if obj.spec.cron_schedule.is_some() {
353            let cronjob = cronjobs::get_cronjob_for_imagesync(&obj, &client).await?;
354            if let Some(existing_cronjob) = cronjob {
355                // CronJob exists, assert that the spec is correct with what's in the CR.
356                if !cronjobs::is_cronjob_spec_correct(&obj, &existing_cronjob, &CONFIG.skopeo).await
357                {
358                    // CronJob spec is incorrect, delete it and recreate it.
359                    cronjobs::delete_cronjob(&existing_cronjob, &client).await?;
360                    cronjobs::create_cronjob(&obj, &CONFIG.skopeo, &client).await?;
361                    // Normal requeue
362                    return Ok(Action::requeue(Duration::from_secs(30)));
363                } else {
364                    // CronJob spec is correct, update the status with the last successful run time.
365                    let last_success = existing_cronjob
366                        .status
367                        .as_ref()
368                        .and_then(|status| status.last_successful_time.clone());
369                    reconciler::update_status(
370                        obj.as_ref().clone(),
371                        true,
372                        true,
373                        String::from("ImageSync CronJob is running"),
374                        last_success,
375                        &client,
376                    )
377                    .await?;
378                    return Ok(Action::requeue(Duration::from_secs(30)));
379                }
380            } else {
381                cronjobs::create_cronjob(&obj, &CONFIG.skopeo, &client).await?;
382                // Standard requeue
383                return Ok(Action::requeue(Duration::from_secs(30)));
384            }
385        } else {
386            // If the CR is ready and does not have a cron schedule, there's nothing to do. Do a very long requeue.
387            return Ok(Action::requeue(Duration::from_secs(3600)));
388        }
389    }
390
391    // If the CR is accepted but not ready, we need to create a one-shot job to process it. We always fire a one-shot, even when the job is scheduled.
392    let job = jobs::get_job_for_imagesync(&obj, &client).await?;
393    // Create job when absent
394    let Some(job) = job else {
395        jobs::create_job(&obj, &CONFIG.skopeo, &client).await?;
396        return Ok(Action::requeue(Duration::from_secs(5)));
397    };
398
399    // If we get here, the job exists. Check its status.
400    if jobs::is_job_complete(&job).await {
401        // Job is not running, check if it succeeded or failed
402        if jobs::is_job_failed(&job).await {
403            // Job failed. Check how long ago the job failed. If it was more than 10 minutes ago we will delete it to retry.
404            let failure_time = job
405                .status
406                .as_ref()
407                .and_then(|status| status.conditions.as_ref())
408                .and_then(|conditions| {
409                    conditions
410                        .iter()
411                        .find(|c| c.type_ == "Failed" && c.status == "True")
412                })
413                .and_then(|condition| condition.last_transition_time.as_ref())
414                .ok_or_else(|| {
415                    OpError::from(format!(
416                        "Failed Job for ImageSync {name} has no failure transition time"
417                    ))
418                })?;
419            if failure_time.0.duration_until(Timestamp::now()).as_mins() > 10 {
420                // Delete the job to retry
421                jobs::delete_job(&job, &client).await?;
422                // Fast requeue to watch the job status
423                reconciler::update_status(
424                    obj.as_ref().clone(),
425                    true,
426                    false,
427                    String::from("ImageSync job failed, retrying"),
428                    None,
429                    &client,
430                )
431                .await?;
432                Ok(Action::requeue(Duration::from_secs(5)))
433            } else {
434                reconciler::update_status(
435                    obj.as_ref().clone(),
436                    true,
437                    false,
438                    String::from("ImageSync job failed"),
439                    None,
440                    &client,
441                )
442                .await?;
443                // Normal requeue to check the job status again
444                Ok(Action::requeue(Duration::from_secs(30)))
445            }
446        } else {
447            // Job succeeded. Update the status to ready.
448            reconciler::update_status(
449                obj.as_ref().clone(),
450                true,
451                true,
452                String::from("ImageSync job completed successfully"),
453                Some(
454                    job.status
455                        .as_ref()
456                        .and_then(|status| status.completion_time.clone())
457                        .ok_or_else(|| {
458                            OpError::from(format!(
459                                "Completed Job for ImageSync {name} has no completion time"
460                            ))
461                        })?,
462                ),
463                &client,
464            )
465            .await?;
466            // Fast requeue since we don't know if we need to create a CronJob or not.
467            Ok(Action::requeue(Duration::from_secs(5)))
468        }
469    } else {
470        // Job is still running, update the status to not ready.
471        reconciler::update_status(
472            obj.as_ref().clone(),
473            true,
474            false,
475            String::from("ImageSync job is running"),
476            None,
477            &client,
478        )
479        .await?;
480        // Fast requeue to watch the job status
481        Ok(Action::requeue(Duration::from_secs(5)))
482    }
483}

485/// This function is called when the reconciler encounters an error. In this case, we simply retry the reconcile after 30 seconds.
486/// Because the reconciler is idempotent, this is safe to do. The error is logged by the controller runtime.
487fn error_policy(_object: Arc<ImageSync>, _err: &OpError, _ctx: Arc<()>) -> Action {
488    eprintln!("Reconcile error: {:?} on {:?}", _err, _object.metadata.name);
489    Action::requeue(Duration::from_secs(30))
